\section{Foundations}
\label{sec:foundations}

\begin{definition}[Subgradient Bundle Condition]
\label{def:subgradient_bundle}
A function $f: \mathbb{R}^d \to \mathbb{R}$ satisfies the $(\alpha, L)$-subgradient bundle condition if for all $x, y \in \text{dom}(f)$, the directional derivative $\nabla f(x; y - x) \ge \alpha \|y - x\|^2 - L \|y - x\|$.
\end{definition}
